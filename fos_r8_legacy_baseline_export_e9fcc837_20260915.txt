def canonical(value):
    raw = json.dumps(
        value, ensure_ascii=False, sort_keys=True, separators=(",", ":")
    ).encode("utf-8")
    return hashlib.sha256(raw).hexdigest()


def sha256_file(path):
    digest = hashlib.sha256()
    with open(path, "rb") as stream:
        for chunk in iter(lambda: stream.read(1024 * 1024), b""):
            digest.update(chunk)
    return digest.hexdigest()


def file_info(path):
    requested = os.path.abspath(str(path))
    if os.path.islink(requested) or not os.path.isfile(requested):
        raise ValueError("regular file required: {}".format(path))
    file_stat = os.stat(requested, follow_symlinks=False)
    return {
        "filename": os.path.basename(requested),
        "path": requested,
        "size": int(file_stat.st_size),
        "sha256": sha256_file(requested),
    }


def resolved_path(path):

def safe_relative(value):
    value = str(value).replace("\\", "/")
    parts = value.split("/")
    return bool(value) and not value.startswith("/") and \
        all(part not in ("", ".", "..") for part in parts)


def _path_within(path, root):
    return path == root or path.startswith(root + os.sep)


def _normalise_export_path(relative, actual_path=None, export_root=None):
    value = str(relative).replace("\\", "/")
    if value.startswith("/"):
        if export_root is None:
            raise ValueError("absolute path is not export-relative")
        root_abs = os.path.abspath(str(export_root))
        value = os.path.relpath(os.path.abspath(value), root_abs).replace(
            os.sep, "/"
        )
    if not safe_relative(value):
        raise ValueError("path is not safely contained: {}".format(relative))
    normalised = posixpath.normpath(value)
    if normalised != value or not safe_relative(normalised):
        raise ValueError("path is not canonical or safely contained: {}".format(
            relative
        ))
    parts = normalised.split("/")
    if actual_path is not None:
        requested = os.path.abspath(str(actual_path))
        if os.path.islink(requested):
            raise ValueError("symlink is not exportable: {}".format(relative))
        if export_root is not None:
            root_real = resolved_path(export_root)
            actual_real = resolved_path(requested)
            if not _path_within(actual_real, root_real):
                raise ValueError("path escapes Served Root: {}".format(relative))
    return normalised, parts


def _is_coverage_source_artifact(parts):
    basename = parts[-1].lower()
    if basename in COVERAGE_HTML_BASENAMES:
        return True
    if SOURCE_COVERAGE_BASENAME_RE.search(basename):
        return True
    if basename.endswith((".html", ".htm")) and any(
            component.lower() in COVERAGE_DIRECTORY_COMPONENTS
            for component in parts[:-1]
    ):
        return True
    return False


def _sensitive_path_reason(parts, is_directory=False):
    lower_parts = [part.lower() for part in parts]
    basename = lower_parts[-1]
    if any(part in PROTECTED_DIRECTORY_COMPONENTS for part in lower_parts):
        return "protected directory component"
    if is_directory:
        return None
    if basename in SENSITIVE_EXACT_BASENAMES:
        return "known sensitive filename"
    if SENSITIVE_EXTENSION_RE.search(basename):
        return "known sensitive file type"
    if ENV_SECRET_FILENAME_RE.match(basename):
        return "environment secret filename"
    if _is_coverage_source_artifact(parts):
        return None
    if PRIVATE_KEY_FILENAME_RE.search(basename):
        return "private-key filename"
    for part in lower_parts:
        if SENSITIVE_PATH_KEYWORD_RE.search(part):
            return "sensitive-looking non-artifact path component"
    return None


def reject_sensitive_path(relative, actual_path=None, export_root=None):
    normalised, parts = _normalise_export_path(
        relative, actual_path=actual_path, export_root=export_root
    )
    is_directory = actual_path is not None and os.path.isdir(actual_path)
    reason = _sensitive_path_reason(parts, is_directory=is_directory)
    if reason:
        raise ValueError(
            "sensitive-looking path is not exported: {} ({})".format(
                normalised, reason
            )
        )


def inventory(root):
    requested = os.path.abspath(str(root))
    if os.path.islink(requested) or not os.path.isdir(requested):
        raise ValueError("root is not a real directory: {}".format(root))
    root = os.path.realpath(requested)
    files = []
    directories = []

    def visit(directory, relative_directory=""):
        try:
            names = sorted(os.listdir(directory))
        except OSError as exc:
            raise ValueError("cannot scan {}: {}".format(directory, exc))
        for name in names:
            path = os.path.join(directory, name)
            relative = name if not relative_directory else \
                relative_directory + "/" + name
            if os.path.islink(path):
                raise ValueError("symlink is not allowed: {}".format(relative))
            try:
                file_stat = os.stat(path, follow_symlinks=False)
            except OSError as exc:
                raise ValueError("cannot stat {}: {}".format(relative, exc))
            if stat.S_ISDIR(file_stat.st_mode):
                reject_sensitive_path(relative, path, root)
                directories.append(relative)
                visit(path, relative)
            elif stat.S_ISREG(file_stat.st_mode):
                reject_sensitive_path(relative, path, root)
                files.append({
                    "path": relative,
                    "size": int(file_stat.st_size),
                    "sha256": sha256_file(path),
                })
            else:
                raise ValueError("special file is not allowed: {}".format(relative))

    visit(root)
    files.sort(key=lambda item: item["path"])
    directories.sort()
    return {
        "files": files,
        "directories": directories,
        "file_count": len(files),
        "total_size": sum(int(item["size"]) for item in files),
        "tree_sha256": canonical(files),
    }


def write_json(path, value):
    parent = os.path.dirname(os.path.abspath(path))
    if parent and not os.path.isdir(parent):
        os.makedirs(parent)
    with open(path, "w", encoding="utf-8") as stream:
        json.dump(value, stream, ensure_ascii=False, indent=2, sort_keys=True)
        stream.write("\n")


def load_json(path):
    with open(path, "r", encoding="utf-8") as stream:
        value = json.load(stream)
    if not isinstance(value, dict):
        raise ValueError("JSON object required: {}".format(path))
    return value


def copy_files(source, destination, skip_controls=False):
    source_inventory = inventory(source)
    source_root = os.path.realpath(os.path.abspath(str(source)))
    destination = os.path.abspath(str(destination))
    if os.path.lexists(destination):
        if os.path.islink(destination) or not os.path.isdir(destination):
            raise ValueError("destination is not a real directory: {}".format(destination))
    else:
        os.makedirs(destination)
    skipped_directories = set()
    for relative in source_inventory["directories"]:
        top_level = relative.split("/", 1)[0]
        if skip_controls and top_level in CONTROL_DIRECTORIES:
            skipped_directories.add(relative)
            continue
        if any(relative == parent or relative.startswith(parent + "/")
               for parent in skipped_directories):
            skipped_directories.add(relative)
            continue
        target_directory = os.path.join(destination, *relative.split("/"))
        if not os.path.lexists(target_directory):
            os.makedirs(target_directory)
        elif os.path.islink(target_directory) or not os.path.isdir(target_directory):
            raise ValueError("destination directory is unsafe: {}".format(relative))
    copied = 0
    skipped = []
    for item in source_inventory["files"]:
        relative = item["path"]
        basename = relative.rsplit("/", 1)[-1]
        top_level = relative.split("/", 1)[0]
        if skip_controls and (top_level in CONTROL_DIRECTORIES or
                              basename in CONTROL_NAMES):
            skipped.append(relative)
            continue
        target = os.path.join(destination, *relative.split("/"))
        target_parent = os.path.dirname(target)
        if not os.path.isdir(target_parent):
            os.makedirs(target_parent)
        if os.path.lexists(target) and os.path.islink(target):
            raise ValueError("destination file is a symlink: {}".format(relative))
        shutil.copy2(os.path.join(source_root, *relative.split("/")), target)
        copied += 1
    return source_inventory, copied, skipped


def compare(left_path, right_path):
    left = load_json(left_path)
    right = load_json(right_path)
    if left.get("files") != right.get("files") or \
            left.get("directories") != right.get("directories"):
        raise ValueError("inventory changed")
    return left


def check_space(path, inventory_path):
    value = load_json(inventory_path)
    stats = os.statvfs(os.path.abspath(str(path)))
    free_bytes = int(stats.f_bavail) * int(stats.f_frsize)
    required_bytes = int(value.get("total_size") or 0) * 2 + 1024 * 1024
    result = {
        "free_bytes": free_bytes,
        "required_bytes": required_bytes,
        "status": "PASS" if free_bytes >= required_bytes else "FAIL",
    }
    if result["status"] != "PASS":
        raise ValueError("insufficient free space for immutable export")
    return result


def detect_identity_mode(identity_path):
    """Select strict existing identity or explicit legacy pre-adoption mode."""
    requested = os.path.abspath(str(identity_path))
    if os.path.lexists(requested):
        if os.path.islink(requested) or not os.path.isfile(requested):
            raise ValueError(
                "release_identity.json is present but is not a regular file"
            )
        return "EXISTING_VERIFIED_IDENTITY"
    return "LEGACY_PRE_ADOPTION"


def detect_application_identity_mode(application_root):
    """Select checkout identity or the explicit packaged-release identity."""
    root = os.path.abspath(str(application_root))
    git_metadata = os.path.join(root, ".git")
    if os.path.lexists(git_metadata):
        return "GIT_CHECKOUT"
    manifest_path = os.path.join(root, "release_manifest.json")
    if os.path.islink(manifest_path) or not os.path.isfile(manifest_path):
        raise ValueError(
            "packaged application requires a regular release_manifest.json"
        )
    return "PACKAGED_RELEASE_MANIFEST"


def _valid_git_object_sha(value):
    value = str(value or "").strip().lower()
    return bool(COMMIT_RE.fullmatch(value)) and set(value) != {"0"}


def _git_value(application_root, arguments, label):
    try:
        value = subprocess.check_output(
            ["git", "-C", os.path.abspath(str(application_root))] + list(arguments),
            stderr=subprocess.DEVNULL,
        ).decode("utf-8", "replace").strip()
    except (OSError, subprocess.CalledProcessError) as exc:
        raise ValueError("{} cannot be verified: {}".format(label, exc))
    if not value or "\n" in value:
        raise ValueError("{} is ambiguous".format(label))
    return value


def validate_git_checkout(application_root, expected_commit):
    """Verify the exact checkout HEAD/tree without inventing package metadata."""
    root = os.path.abspath(str(application_root))
    if os.path.islink(root) or not os.path.isdir(root):
        raise ValueError("application root is not a real directory")
    if not os.path.lexists(os.path.join(root, ".git")):
        raise ValueError("Git checkout metadata is absent")
    top_level = _git_value(root, ["rev-parse", "--show-toplevel"],
                           "Git checkout root")
    if os.path.realpath(top_level) != os.path.realpath(root):
        raise ValueError("Git checkout root does not match application root")
    head = _git_value(root, ["rev-parse", "HEAD"], "Git HEAD").lower()
    tree = _git_value(root, ["rev-parse", "HEAD^{tree}"], "Git tree").lower()
    expected = str(expected_commit or "").strip().lower()
    if not COMMIT_RE.fullmatch(expected):
        raise ValueError("expected Git baseline SHA is not exact")
    if not _valid_git_object_sha(head) or head != expected:
        raise ValueError("Git HEAD does not match expected baseline")
    if not _valid_git_object_sha(tree):
        raise ValueError("Git tree SHA is not exact")
    return {
        "application_identity_mode": "GIT_CHECKOUT",
        "application_git_sha": head,
        "application_git_tree_sha": tree,
        "packaged_release_anchor": None,
    }


def _packaged_asset_path(application_root, relative):
    root = os.path.abspath(str(application_root))
    root_real = os.path.realpath(root)
    parts = relative.split("/")
    candidate = os.path.join(root, *parts)
    candidate_real = os.path.realpath(candidate)
    if not _path_within(candidate_real, root_real):
        raise ValueError("packaged asset escapes application root: {}".format(
            relative
        ))
    current = root
    for part in parts[:-1]:
        current = os.path.join(current, part)
        if os.path.islink(current) or not os.path.isdir(current):
            raise ValueError("packaged asset parent is not a real directory: {}".format(
                relative
            ))
    if os.path.islink(candidate) or not os.path.lexists(candidate):
        raise ValueError("packaged asset is missing or linked: {}".format(relative))
    file_stat = os.stat(candidate, follow_symlinks=False)
    if not stat.S_ISREG(file_stat.st_mode):
        raise ValueError("packaged asset is not a regular file: {}".format(relative))
    return candidate


def _validate_packaged_asset_manifest(declared, application_root):
    if not isinstance(declared, list) or not declared:
        raise ValueError("release manifest asset_manifest is invalid")
    normalized = []
    seen = set()
    root = os.path.abspath(str(application_root))
    for item in declared:
        if not isinstance(item, dict) or set(item) != set(("path", "size", "sha256")):
            raise ValueError("release manifest asset record is invalid")
        path = item.get("path")
        if not isinstance(path, str) or path != path.replace("\\", "/") or \
                not safe_relative(path) or posixpath.normpath(path) != path:
            raise ValueError("release manifest asset path is invalid: {}".format(
                path
            ))
        if path in seen:
            raise ValueError("release manifest has duplicate asset path: {}".format(
                path
            ))
        seen.add(path)
        size = item.get("size")
        if not isinstance(size, int) or isinstance(size, bool):
            raise ValueError("release manifest asset size is invalid: {}".format(
                path
            ))
        digest = item.get("sha256")
        if not isinstance(digest, str) or digest != digest.lower() or \
                size < 0 or not SHA256_RE.fullmatch(digest):
            raise ValueError("release manifest asset fingerprint is invalid: {}".format(
                path
            ))
        normalized.append({"path": path, "size": size, "sha256": digest})
    if normalized != sorted(normalized, key=lambda value: value["path"]):
        raise ValueError("release manifest asset_manifest is not canonical")
    for item in normalized:
        path = _packaged_asset_path(root, item["path"])
        file_stat = os.stat(path, follow_symlinks=False)
        observed_hash = sha256_file(path)
        if int(file_stat.st_size) != item["size"] or \
                observed_hash.lower() != item["sha256"]:
            raise ValueError("packaged asset bytes do not match: {}".format(
                item["path"]
            ))
    return normalized


def validate_packaged_release_manifest(
        manifest_path, application_root, expected_commit=None,
        expected_manifest_sha256=None, expected_asset_hash=None):
    """Verify a packaged release manifest and every declared package byte."""
    root = os.path.abspath(str(application_root))
    if os.path.islink(root) or not os.path.isdir(root):
        raise ValueError("application root is not a real directory")
    expected_manifest_path = os.path.join(root, "release_manifest.json")
    requested_manifest = os.path.abspath(str(manifest_path))
    if requested_manifest != expected_manifest_path:
        raise ValueError("packaged release manifest path is not at application root")
    manifest_info = file_info(requested_manifest)
    expected_commit = str(
        expected_commit if expected_commit is not None else CONTRACT["baseline_sha"]
    ).strip().lower()
    expected_manifest_sha256 = str(
        expected_manifest_sha256
        if expected_manifest_sha256 is not None
        else CONTRACT["known_release_manifest_sha256"]
    ).strip().lower()
    expected_asset_hash = str(
        expected_asset_hash
        if expected_asset_hash is not None
        else CONTRACT["known_release_asset_hash"]
    ).strip().lower()
    if not COMMIT_RE.fullmatch(expected_commit):
        raise ValueError("expected packaged baseline SHA is not exact")
    if not SHA256_RE.fullmatch(expected_manifest_sha256):
        raise ValueError("expected release manifest SHA256 is malformed")
    if not SHA256_RE.fullmatch(expected_asset_hash):
        raise ValueError("expected release asset hash is malformed")
    if manifest_info["sha256"].lower() != expected_manifest_sha256:
        raise ValueError("release manifest file SHA256 does not match known release")
    manifest = load_json(requested_manifest)
    for key in EXPECTED_IDENTITY_KEYS:
        if manifest.get(key) in (None, ""):
            raise ValueError("release manifest is incomplete: {}".format(key))
    version = manifest.get("version")
    if not isinstance(version, str) or not version.strip():
        raise ValueError("release manifest version is invalid")
    commit_sha = manifest.get("commit_sha")
    if not isinstance(commit_sha, str) or commit_sha != commit_sha.lower() or \
            not COMMIT_RE.fullmatch(commit_sha) or commit_sha != expected_commit:
        raise ValueError("release manifest commit_sha does not match baseline")
    schema_version = manifest.get("schema_version")
    asset_manifest_version = manifest.get("asset_manifest_version")
    if not isinstance(schema_version, int) or isinstance(schema_version, bool) or \
            schema_version != EXPECTED_RELEASE_SCHEMA_VERSION:
        raise ValueError("release manifest schema_version is invalid")
    if not isinstance(asset_manifest_version, int) or \
            isinstance(asset_manifest_version, bool) or \
            asset_manifest_version != EXPECTED_ASSET_MANIFEST_VERSION:
        raise ValueError("release manifest asset_manifest_version is invalid")
    declared = _validate_packaged_asset_manifest(
        manifest.get("asset_manifest"), root
    )
    asset_count = manifest.get("asset_count")
    if not isinstance(asset_count, int) or isinstance(asset_count, bool):
        raise ValueError("release manifest asset_count is invalid")
    if asset_count != len(declared):
        raise ValueError("release manifest asset_count does not match assets")
    asset_manifest_hash = canonical(declared)
    manifest_asset_manifest_hash = manifest.get("asset_manifest_hash")
    if not isinstance(manifest_asset_manifest_hash, str) or \
            manifest_asset_manifest_hash != asset_manifest_hash:
        raise ValueError("release manifest asset_manifest_hash is invalid")
    manifest_asset_hash = manifest.get("asset_hash")
    if not isinstance(manifest_asset_hash, str) or \
            manifest_asset_hash != asset_manifest_hash:
        raise ValueError("release manifest asset_hash is invalid")
    if asset_manifest_hash != expected_asset_hash:
        raise ValueError("release manifest asset hash does not match known release")
    build_id = str(manifest.get("build_id") or "")
    expected_build_id = "{}-{}-{}".format(
        version.split()[0], commit_sha[:8], asset_manifest_hash[:8]
    )
    if build_id != expected_build_id:
        raise ValueError("release manifest build_id is invalid")
    anchor = {
        "anchor_type": "PACKAGED_RELEASE_MANIFEST",
        "verified": True,
        "release_manifest_path": requested_manifest,
        "release_manifest_sha256": manifest_info["sha256"].lower(),
        "commit_sha": commit_sha,
        "asset_hash": asset_manifest_hash,
        "asset_manifest_hash": asset_manifest_hash,
        "asset_count": asset_count,
        "schema_version": int(schema_version),
        "asset_manifest_version": int(asset_manifest_version),
        "version": version,
        "build_id": build_id,
    }
    return {
        "application_identity_mode": "PACKAGED_RELEASE_MANIFEST",
        "application_git_sha": APPLICATION_GIT_UNAVAILABLE,
        "application_git_tree_sha": APPLICATION_GIT_UNAVAILABLE,
        "release_manifest_path": requested_manifest,
        "release_manifest_sha256": manifest_info["sha256"].lower(),
        "release_manifest_file_sha256": manifest_info["sha256"].lower(),
        "release_manifest_asset_hash": asset_manifest_hash,
        "release_manifest_asset_manifest_hash": asset_manifest_hash,
        "release_manifest_asset_count": asset_count,
        "release_manifest_schema_version": int(schema_version),
        "release_manifest_asset_manifest_version": int(asset_manifest_version),
        "release_manifest_version": version,
        "release_manifest_build_id": build_id,
        "packaged_release_anchor": anchor,
        "packaged_asset_bindings": declared,
    }


def validate_application_identity(
        application_root, expected_commit=None, expected_manifest_sha256=None,
        expected_asset_hash=None):
    mode = detect_application_identity_mode(application_root)
    if mode == "GIT_CHECKOUT":
        return validate_git_checkout(
            application_root,
            expected_commit if expected_commit is not None else CONTRACT["baseline_sha"],
        )
    return validate_packaged_release_manifest(
        os.path.join(os.path.abspath(str(application_root)), "release_manifest.json"),
        application_root,
        expected_commit=expected_commit,
        expected_manifest_sha256=expected_manifest_sha256,
        expected_asset_hash=expected_asset_hash,
    )


def _identity_manifest_from_inventory(value, description):
    files = value.get("files")
    if not isinstance(files, list) or not files:
        raise ValueError("{} has no regular files".format(description))
    declared = []
    seen = set()
    for item in files:
        if not isinstance(item, dict):
            raise ValueError("{} inventory record is invalid".format(description))
        path = str(item.get("path") or "").replace("\\", "/")
        if not safe_relative(path):
            raise ValueError("{} path is invalid: {}".format(description, path))
        if path in seen:
            raise ValueError("{} has duplicate path: {}".format(
                description, path
            ))
        seen.add(path)
        try:
            size = int(item.get("size"))
        except (TypeError, ValueError):
            raise ValueError("{} size is invalid: {}".format(description, path))
        digest = str(item.get("sha256") or "").lower()
        if size < 0 or not SHA256_RE.fullmatch(digest):
            raise ValueError("{} fingerprint is invalid: {}".format(
                description, path
            ))
        declared.append({"path": path, "size": size, "sha256": digest})
    declared.sort(key=lambda item: item["path"])
    try:
        file_count = int(value.get("file_count"))
        total_size = int(value.get("total_size"))
    except (TypeError, ValueError):
        raise ValueError("{} summary is invalid".format(description))
    if file_count != len(declared) or total_size != sum(
            item["size"] for item in declared):
        raise ValueError("{} summary does not match files".format(description))
    tree_sha = str(value.get("tree_sha256") or "").lower()
    if not SHA256_RE.fullmatch(tree_sha) or canonical(declared) != tree_sha:
        raise ValueError("{} tree identity is invalid".format(description))
    return declared

    snapshot = load_json(snapshot_inventory_path)
    adoption = load_json(adoption_inventory_path)
    expected_commit = str(expected_commit or "").lower()
    if not COMMIT_RE.fullmatch(expected_commit):
        raise ValueError("expected legacy baseline SHA is not exact")

        if isinstance(application_identity, str):
            application_identity = load_json(application_identity)
        if not isinstance(application_identity, dict):

        app_git_tree = str(app_git_tree or "").lower()
        if app_git_sha != expected_commit or not _valid_git_object_sha(app_git_sha):
            raise ValueError("application Git baseline does not match expected baseline")
        if not _valid_git_object_sha(app_git_tree):
            raise ValueError("application Git tree SHA is not exact")
        packaged_anchor = None
    _identity_manifest_from_inventory(snapshot, "legacy snapshot inventory")
    declared = _identity_manifest_from_inventory(
        adoption, "legacy adoption inventory"
    )
    snapshot_tree = str(snapshot.get("tree_sha256") or "").lower()
    adoption_tree = str(adoption.get("tree_sha256") or "").lower()
    asset_hash = canonical(declared)
    if application_mode == "PACKAGED_RELEASE_MANIFEST":

        "asset_count": len(declared),
        "asset_manifest_hash": asset_hash,
        "asset_manifest": declared,
        "schema_version": 2,
        "build_provenance": provenance,
        "built_at": str(timestamp),
        "identity_mode": "LEGACY_PRE_ADOPTION",
        "identity_origin": "legacy_adoption",
        "original_release_identity": "ABSENT",
        "historical_build_attestation": "NOT_AVAILABLE",
        "provenance_claim": provenance_claim,
        "expected_baseline_sha": expected_commit,
        "application_git_sha": app_git_sha,

def validate_identity(identity_path, source_root, expected_commit, normalized_path):
    identity = load_json(identity_path)
    if "release_identity" in identity:
        raise ValueError("identity must be a direct object")
    for key in identity:
        if SENSITIVE_METADATA_KEY_RE.search(str(key)):
            raise ValueError("identity contains a sensitive-looking key")
    missing = [key for key in EXPECTED_IDENTITY_KEYS
               if identity.get(key) in (None, "")]
    if missing:
        raise ValueError("identity is incomplete: {}".format(",".join(missing)))
    commit_sha = str(identity.get("commit_sha") or "").lower()
    if not COMMIT_RE.fullmatch(commit_sha):
        raise ValueError("identity commit_sha is not exact")
    if commit_sha != str(expected_commit).lower():
        raise ValueError("identity commit_sha does not match fixed baseline")
    declared = identity.get("asset_manifest")
    if not isinstance(declared, list) or not declared:
        raise ValueError("identity asset_manifest is invalid")
    if int(identity.get("asset_count") or 0) != len(declared):
        raise ValueError("identity asset_count is invalid")
    if int(identity.get("asset_manifest_version") or 0) <= 0:
        raise ValueError("identity asset_manifest_version is invalid")
    if canonical(declared) != str(identity.get("asset_hash") or "").lower():
        raise ValueError("identity asset_hash is invalid")
    if canonical(declared) != str(identity.get("asset_manifest_hash") or "").lower():
        raise ValueError("identity asset_manifest_hash is invalid")
    source = inventory(source_root)
    by_path = dict((item["path"], item) for item in source["files"])
    by_basename = {}
    for item in source["files"]:
        if "/" not in item["path"]:
            by_basename.setdefault(item["path"], []).append(item)
    bindings = []
    for item in sorted(declared, key=lambda value: str(value.get("path", ""))):
        path = str(item.get("path") or "").replace("\\", "/")
        if not safe_relative(path):
            raise ValueError("identity asset path is invalid: {}".format(path))
        try:
            expected_size = int(item.get("size"))
        except (TypeError, ValueError):
            raise ValueError("identity asset size is invalid: {}".format(path))
        expected_hash = str(item.get("sha256") or "").lower()
        if expected_size < 0 or not SHA256_RE.fullmatch(expected_hash):
            raise ValueError("identity asset fingerprint is invalid: {}".format(path))
        observed = by_path.get(path)
        if observed is None:
            candidates = by_basename.get(os.path.basename(path), [])
            if len(candidates) != 1:
                raise ValueError("identity asset is missing or ambiguous: {}".format(path))
            observed = candidates[0]
        observed_path = os.path.join(
            os.path.realpath(os.path.abspath(str(source_root))),
            *observed["path"].split("/")
        )
        try:
            reject_sensitive_path(path, observed_path, source_root)
        except ValueError:
            raise ValueError("identity asset path is sensitive-looking: {}".format(
                path
            ))
        if int(observed["size"]) != expected_size or \
                observed["sha256"].lower() != expected_hash:
            raise ValueError("identity asset bytes do not match: {}".format(path))
        bindings.append({
            "identity_path": path,
            "source_path": observed["path"],
            "size": expected_size,
            "sha256": expected_hash,
        })
    allowed = set(EXPECTED_IDENTITY_KEYS)
    allowed.update(("project_name", "report_mode", "release_mode"))
    normalized = dict((key, identity[key]) for key in identity if key in allowed)
    write_json(normalized_path, normalized)
    return {
        "identity_sha256": canonical(identity),
        "identity_file_sha256": sha256_file(identity_path),
        "commit_sha": commit_sha,
        "asset_count": len(declared),
        "asset_bindings": bindings,
        "normalized_path": os.path.abspath(normalized_path),
    }


def redact_line(line):
    def redact_uri(match):
        scheme = match.group(1)
        userinfo = match.group(2)
        value = userinfo[:-1]
        username = value.split(":", 1)[0]
        return scheme + username + ":REDACTED@"

    line = URI_RE.sub(redact_uri, line)
    line = re.sub(
        r"(?i)(\b(?:password|passwd|secret|token|hmac|private[._-]?key|"
        r"authorization|cookie|credential|api[._-]?key)\b\s*[=:]\s*)"
        r"([^\s,;]+)",
        r"\1REDACTED",
        line,
    )
    line = re.sub(
        r"(?i)(--password(?:=|\s+)|--secret(?:=|\s+))([^\s]+)",
        r"\1REDACTED",
        line,
    )
    return line.rstrip()


def sanitize_file(input_path, output_path):
    with open(input_path, "r", encoding="utf-8", errors="replace") as source:
        lines = [redact_line(line) for line in source]
    with open(output_path, "w", encoding="utf-8") as target:
        for line in lines:
            target.write(line + "\n")


def sidecar_summary(root, output_path):
    root = os.path.realpath(os.path.abspath(str(root)))
    cache_root = os.path.join(root, "reports", ".source_cache")
    records = []
    invalid = []
    if os.path.isdir(cache_root):
        for directory, dirnames, filenames in os.walk(
                cache_root, topdown=True, followlinks=False):
            dirnames[:] = sorted(dirnames)
            filenames = sorted(filenames)
            for name in filenames:
                if name != "meta.json":
                    continue
                path = os.path.join(directory, name)
                relative = os.path.relpath(path, root).replace(os.sep, "/")
                if os.path.islink(path) or not os.path.isfile(path):
                    invalid.append(relative + ":not-a-regular-file")
                    continue
                try:
                    value = load_json(path)
                except (IOError, OSError, ValueError, TypeError) as exc:
                    invalid.append(relative + ":invalid-json:" + str(exc))
                    continue
                schema = value.get("schema_version")
                if isinstance(schema, bool):
                    schema = None
                try:
                    schema = int(schema)
                except (TypeError, ValueError):
                    schema = None
                mode = str(value.get("report_mode") or "").upper()
                mode = re.sub(r"[^A-Z0-9_.-]", "_", mode) or "ABSENT"
                total_lines = value.get("total_lines")
                has_total_lines = (
                    isinstance(total_lines, int) and
                    not isinstance(total_lines, bool) and total_lines >= 0
                )
                ranges = value.get("function_ranges")
                has_function_ranges = isinstance(ranges, list)
                if schema is None:
                    invalid.append(relative + ":schema_version-missing")
                if schema is not None and schema >= 2 and \
                        (not has_total_lines or not has_function_ranges):
                    invalid.append(relative + ":schema-v2-fields-missing")
                records.append({
                    "path": relative,
                    "schema_version": schema if schema is not None else "NOT_VERIFIED",
                    "report_mode": mode,
                    "total_lines": "PRESENT" if has_total_lines else "ABSENT",
                    "function_ranges": (
                        "PRESENT" if has_function_ranges else "ABSENT"
                    ),
                    "sha256": sha256_file(path),
                })
    records.sort(key=lambda item: item["path"])
    schema_v2 = [item for item in records if item["schema_version"] == 2]
    schema_v1 = [item for item in records if item["schema_version"] == 1]
    modes = sorted(set(item["report_mode"] for item in records))
    if invalid:
        status = "FAIL"
    elif records:
        status = "PASS"
    else:
        status = "NOT_APPLICABLE"
    result = {
        "SIDECAR_LAYOUT": "reports/.source_cache/**/meta.json",
        "SIDECAR_STATUS": status,
        "SIDECAR_META_JSON_COUNT": len(records),
        "SIDECAR_SCHEMA_V2_COUNT": len(schema_v2),
        "SIDECAR_SCHEMA_V1_COUNT": len(schema_v1),
        "SIDECAR_V2_TOTAL_LINES_COUNT": sum(
            1 for item in schema_v2 if item["total_lines"] == "PRESENT"
        ),
        "SIDECAR_V2_FUNCTION_RANGES_COUNT": sum(
            1 for item in schema_v2 if item["function_ranges"] == "PRESENT"
        ),
        "SIDECAR_REPORT_MODES": ",".join(modes) if modes else "ABSENT",
        "SIDECAR_INVALID": "|".join(invalid) if invalid else "NONE",
        "SIDECAR_RECORDS": json.dumps(
            records, ensure_ascii=False, sort_keys=True,
            separators=(",", ":")
        ) if records else "[]",
    }
    with open(output_path, "w", encoding="utf-8") as stream:
        for key in sorted(result):
            stream.write("{}={}\n".format(key, result[key]))
    if invalid:
        raise ValueError("canonical Sidecar metadata is invalid")
    return result


def _nginx_clean(text):
    return "\n".join(line.split("#", 1)[0] for line in text.splitlines())


def _nginx_sections(raw):
    sections = []
    current = None
    marker = re.compile(
        r"^\s*#\s*configuration file\s+(.+?):\s*$", re.IGNORECASE
    )
    for line in raw.splitlines():
        match = marker.match(line)
        if match:
            current = {"path": match.group(1).strip(), "lines": []}
            sections.append(current)
        elif current is not None:
            current["lines"].append(line)
    return sections


def _nginx_block_end(text, opening):
    depth = 0
    quote = None
    escaped = False
    for index in range(opening, len(text)):
        char = text[index]
        if quote is not None:
            if escaped:
                escaped = False
            elif char == "\\":
                escaped = True
            elif char == quote:
                quote = None
            continue
        if char in ("\"", "'"):
            quote = char
        elif char == "{":
            depth += 1
        elif char == "}":
            depth -= 1
            if depth == 0:
                return index
            if depth < 0:
                break
    raise ValueError("nginx block has unbalanced braces")


def _nginx_depth_before(text, position):
    depth = 0
    quote = None
    escaped = False
    for char in text[:position]:
        if quote is not None:
            if escaped:
                escaped = False
            elif char == "\\":
                escaped = True
            elif char == quote:
                quote = None
            continue
        if char in ("\"", "'"):
            quote = char
        elif char == "{":
            depth += 1
        elif char == "}":
            depth -= 1
    if quote is not None or depth < 0:
        raise ValueError("nginx braces or quotes are unbalanced")
    return depth


def _nginx_directives(text):
    """Return only directives at the block's direct nesting level."""
    result = []
    depth = 0
    for raw_line in text.splitlines():
        line = raw_line.strip()
        if depth == 0:
            match = re.match(r"(?i)^([a-z_][a-z0-9_]*)\s+([^;]+);\s*$", line)
            if match:
                result.append((match.group(1).lower(), match.group(2).strip()))
        depth += line.count("{") - line.count("}")
        if depth < 0:
            raise ValueError("nginx block has unbalanced braces")
    if depth != 0:
        raise ValueError("nginx block has unbalanced braces")
    return result


def _nginx_locations(server_body):
    locations = []
    pattern = re.compile(
        r"(?mi)^\s*location\s+(?:(=|\^~|~\*?|~)\s+)?"
        r"([^\s{]+)\s*\{"
    )
    for match in pattern.finditer(server_body):
        if _nginx_depth_before(server_body, match.start()) != 0:
            continue
        end = _nginx_block_end(server_body, match.end() - 1)
        locations.append({
            "modifier": match.group(1) or "",
            "path": match.group(2),
            "body": server_body[match.end():end],
        })
    return locations


def _nginx_directive_values(directives, name):
    return [value for directive, value in directives if directive == name]


def nginx_summary(
        config_path, expected_config, expected_listen, expected_server_name,
        expected_static_location, expected_static_alias, expected_api_location,
        expected_api_proxy, output_path):
    with open(config_path, "r", encoding="utf-8", errors="replace") as stream:
        raw = stream.read()
    sections = _nginx_sections(raw)
    matching_sections = [
        section for section in sections if section["path"] == str(expected_config)
    ]
    if len(matching_sections) != 1:
        raise ValueError(
            "expected production nginx configuration section is missing or duplicated"
        )
    section_text = _nginx_clean("\n".join(matching_sections[0]["lines"]))
    server_pattern = re.compile(r"(?mi)^\s*server\s*\{")
    servers = []
    for match in server_pattern.finditer(section_text):
        if _nginx_depth_before(section_text, match.start()) != 0:
            continue
        end = _nginx_block_end(section_text, match.end() - 1)
        body = section_text[match.end():end]
        directives = _nginx_directives(body)
        servers.append({
            "body": body,
            "listen": _nginx_directive_values(directives, "listen"),
            "server_name": _nginx_directive_values(directives, "server_name"),
        })
    production_servers = [
        server for server in servers
        if server["listen"] == [str(expected_listen)] and
        server["server_name"] == [str(expected_server_name)]
    ]
    if len(production_servers) != 1:
        raise ValueError(
            "production nginx server is missing, ambiguous, or duplicated"
        )
    production = production_servers[0]
    locations = _nginx_locations(production["body"])
    static_locations = [
        location for location in locations
        if location["path"] == str(expected_static_location)
    ]
    api_locations = [
        location for location in locations
        if location["path"] == str(expected_api_location)
    ]
    if len(static_locations) != 1:
        raise ValueError("production static location is missing or duplicated")
    if len(api_locations) != 1:
        raise ValueError("production API location is missing or duplicated")
    static = static_locations[0]
    api = api_locations[0]
    if static["modifier"] or api["modifier"]:
        raise ValueError("production route location modifier is ambiguous")
    static_directives = _nginx_directives(static["body"])
    aliases = _nginx_directive_values(static_directives, "alias")
    roots = _nginx_directive_values(static_directives, "root")
    if len(aliases) != 1 or roots or aliases[0].strip() != str(expected_static_alias):
        raise ValueError("production static alias does not match exact contract")
    api_directives = _nginx_directives(api["body"])
    proxies = _nginx_directive_values(api_directives, "proxy_pass")
    if len(proxies) != 1 or proxies[0].strip() != str(expected_api_proxy):
        raise ValueError("production API proxy_pass does not match exact contract")
    proxy_match = re.fullmatch(
        r"([a-z][a-z0-9+.-]*://[^/]+)(/.*)", str(expected_api_proxy)
    )
    if not proxy_match:
        raise ValueError("expected API proxy contract is malformed")
    result = {
        "nginx_dump_sha256": sha256_file(config_path),
        "nginx_config_path": str(expected_config),
        "nginx_config_paths": str(expected_config),
        "nginx_production_config_scoped": "MATCH",
        "nginx_candidate_gateway_ignored": "MATCH",
        "production_listen": str(expected_listen),
        "production_server_name": str(expected_server_name),
        "production_server_count": 1,
        "production_static_location": str(expected_static_location),
        "production_static_alias": aliases[0].strip(),
        "production_static_alias_match": "MATCH",
        "production_api_location": str(expected_api_location),
        "production_api_proxy_origin": proxy_match.group(1),
        "production_api_proxy_path": proxy_match.group(2),
        "production_api_proxy": proxies[0].strip(),
        "production_api_proxy_match": "MATCH",
        # Preserve the original evidence names for downstream readers.
        "proxy_pass": proxies[0].strip(),
        "proxy_pass_match": "MATCH",
        "coverage_location_count": 1,
        "coverage_prefix": str(expected_static_location),
        "coverage_static_root": aliases[0].strip().rstrip("/"),
        "coverage_static_root_match": "MATCH",
        "coverage_static_root_directive": "alias",
    }
    with open(output_path, "w", encoding="utf-8") as stream:
        for key in sorted(result):
            stream.write("{}={}\n".format(key.upper(), result[key]))
    return result


def systemd_summary(input_path, output_path):
    properties = {}
    with open(input_path, "r", encoding="utf-8", errors="replace") as stream:
        for raw_line in stream:
            line = raw_line.rstrip("\n")
            if "=" not in line:
                continue
            key, value = line.split("=", 1)
            properties[key] = value
    selected = (
        "Id", "LoadState", "ActiveState", "SubState", "UnitFileState",
        "FragmentPath", "User", "Group", "WorkingDirectory", "ExecStart",
        "MainPID",
    )
    with open(output_path, "w", encoding="utf-8") as stream:
        for key in selected:
            if key in properties:
                stream.write("{}={}\n".format(key, redact_line(properties[key])))
            else:
                stream.write("{}=ABSENT\n".format(key))
        raw_environment_files = properties.get("EnvironmentFiles", "")
        paths = []
        for token in raw_environment_files.split():
            if token.startswith("-/"):
                token = token[1:]
            if token.startswith("/"):
                paths.append(token.split("(", 1)[0])
        if paths:
            stream.write("EnvironmentFiles={}\n".format(",".join(sorted(set(paths)))))
        else:
            stream.write("EnvironmentFiles=ABSENT\n")
        if "Environment" in properties:
            names = []
            for token in properties["Environment"].split():
                if "=" in token:
                    names.append(token.split("=", 1)[0])
            stream.write("EnvironmentNames={}\n".format(",".join(sorted(set(names)))))
        else:
            stream.write("EnvironmentNames=ABSENT\n")


def parse_env_file(path, values):
    try:
        with open(path, "r", encoding="utf-8", errors="replace") as stream:
            for raw_line in stream:
                line = raw_line.strip()
                if not line or line.startswith("#") or "=" not in line:
                    continue
                key, value = line.split("=", 1)
                key = key.strip()
                value = value.strip().strip("\"'")
                if key:
                    values[key] = value
    except (IOError, OSError):
        return False
    return True


def systemd_environment_paths(path):
    properties = {}
    with open(path, "r", encoding="utf-8", errors="replace") as stream:
        for raw_line in stream:
            if "=" in raw_line:
                key, value = raw_line.rstrip("\n").split("=", 1)
                properties[key] = value
    paths = []
    for token in properties.get("EnvironmentFiles", "").split():
        if token.startswith("-/"):
            token = token[1:]
        if token.startswith("/"):
            paths.append(token.split("(", 1)[0])
    return sorted(set(paths))


def first_value(values, names):
    for name in names:
        if values.get(name) not in (None, ""):
            return values[name]
    return ""


def db_probe(systemd_path, output_path):
    values = {}
    paths = systemd_environment_paths(systemd_path)
    readable = []
    for path in paths:
        if parse_env_file(path, values):
            readable.append(path)
    systemd_properties = {}
    with open(systemd_path, "r", encoding="utf-8", errors="replace") as stream:
        for raw_line in stream:
            if "=" in raw_line:
                key, value = raw_line.rstrip("\n").split("=", 1)
                systemd_properties[key] = value
    for token in systemd_properties.get("Environment", "").split():
        if "=" in token:
            key, value = token.split("=", 1)
            if key:
                values[key] = value
    host = first_value(values, ("COVERAGE_MYSQL_HOST", "MYSQL_HOST", "DB_HOST"))
    port = first_value(values, ("COVERAGE_MYSQL_PORT", "MYSQL_PORT", "DB_PORT"))
    user = first_value(values, ("COVERAGE_MYSQL_USER", "MYSQL_USER", "DB_USER"))
    database = first_value(
        values, ("COVERAGE_MYSQL_DATABASE", "MYSQL_DATABASE", "DB_NAME", "DATABASE")
    )
    password = first_value(
        values, ("COVERAGE_MYSQL_PASSWORD", "MYSQL_PASSWORD", "MYSQL_PWD",
                 "DB_PASSWORD", "DATABASE_PASSWORD")
    )
    if not host:
        host = "127.0.0.1"
    if not port:
        port = "3306"
    client = shutil.which("mariadb") or shutil.which("mysql")
    result = {
        "DB_ENVIRONMENT_FILES": ",".join(readable) if readable else "ABSENT",
        "DB_CLIENT": client or "ABSENT",
        "DB_CONFIG_HOST": redact_line(host),
        "DB_CONFIG_PORT": redact_line(port),
        "DB_CONFIG_USER": redact_line(user) if user else "ABSENT",
        "DB_CONFIG_DATABASE": redact_line(database) if database else "ABSENT",
        "DB_PASSWORD": "PRESENT" if password else "ABSENT",
    }
    if not client or not user or not database:
        result["DB_STATUS"] = "NOT_VERIFIED"
    else:
        command = [
            client, "--protocol=tcp", "--host", host, "--port", str(port),
            "--user", user, "--database", database, "--batch", "--raw",
            "--skip-column-names",
        ]
        environment = {"PATH": os.environ.get("PATH", ""), "LANG": "C"}
        if password:
            environment["MYSQL_PWD"] = password
        queries = (
            "SELECT VERSION(), CURRENT_USER(), DATABASE();",
            "SELECT COUNT(*) FROM information_schema.tables WHERE table_schema=DATABASE();",
            "SELECT COUNT(*) FROM information_schema.columns WHERE table_schema=DATABASE();",
            "SELECT COUNT(*) FROM information_schema.statistics WHERE table_schema=DATABASE();",
            "SELECT COUNT(*) FROM information_schema.referential_constraints WHERE constraint_schema=DATABASE();",
            "SELECT COUNT(*) FROM information_schema.triggers WHERE trigger_schema=DATABASE();",
        )
        outputs = []
        failed = False
        for query in queries:
            try:
                value = subprocess.check_output(
                    command + ["-e", query], stderr=subprocess.DEVNULL,
                    env=environment, timeout=15,
                ).decode("utf-8", "replace").strip()
                outputs.append(redact_line(value))
            except (OSError, subprocess.CalledProcessError, subprocess.TimeoutExpired):
                failed = True
                outputs.append("NOT_VERIFIED")
        result["DB_STATUS"] = "NOT_VERIFIED" if failed else "PASS"
        result["DB_READ_ONLY_RESULTS"] = "|".join(outputs)
    with open(output_path, "w", encoding="utf-8") as stream:
        for key in sorted(result):
            stream.write("{}={}\n".format(key, result[key]))
    return result


def manifest(export_root, output_path, metadata_path):
    export_root = os.path.realpath(os.path.abspath(str(export_root)))
    metadata = load_json(metadata_path)
    files = []
    directories = []
    excluded = os.path.basename(os.path.abspath(output_path))
    before_path = os.path.join(
        export_root, "evidence", "inventory-source-before.json"
    )
    after_path = os.path.join(
        export_root, "evidence", "inventory-source-after.json"
    )
    copied_path = os.path.join(
        export_root, "evidence", "inventory-legacy-flat-root.json"
    )
    for path in (before_path, after_path, copied_path):
        if not os.path.isfile(path):
            raise ValueError("required inventory evidence is missing: {}".format(path))
    source_before = load_json(before_path)
    source_after = load_json(after_path)
    copied_snapshot = load_json(copied_path)
    sidecar_path = os.path.join(
        export_root, "evidence", "sidecar-summary.txt"
    )
    if not os.path.isfile(sidecar_path):
        raise ValueError("required Sidecar summary is missing: {}".format(
            sidecar_path
        ))
    sidecar = {}
    with open(sidecar_path, "r", encoding="utf-8", errors="replace") as stream:
        for raw_line in stream:
            if "=" in raw_line:
                key, value = raw_line.rstrip("\n").split("=", 1)
                sidecar[key] = value
    if sidecar.get("SIDECAR_LAYOUT") != "reports/.source_cache/**/meta.json":
        raise ValueError("Sidecar layout summary is not canonical")
    if sidecar.get("SIDECAR_STATUS") not in (
            "PASS", "NOT_APPLICABLE", "DEFERRED"):
        raise ValueError("Sidecar status is invalid")
    for key in (
            "SIDECAR_META_JSON_COUNT", "SIDECAR_SCHEMA_V2_COUNT",
            "SIDECAR_SCHEMA_V1_COUNT", "SIDECAR_V2_TOTAL_LINES_COUNT",
            "SIDECAR_V2_FUNCTION_RANGES_COUNT"):
        try:
            if int(sidecar.get(key, "")) < 0:
                raise ValueError
        except (TypeError, ValueError):
            raise ValueError("Sidecar count is invalid: {}".format(key))
    if sidecar.get("SIDECAR_STATUS") == "PASS":
        if sidecar.get("SIDECAR_INVALID") not in ("NONE", ""):
            raise ValueError("passing Sidecar summary contains invalid records")
    if source_before.get("files") != source_after.get("files") or \
            source_before.get("directories") != source_after.get("directories"):
        raise ValueError("source-before/source-after inventory mismatch")
    if source_before.get("files") != copied_snapshot.get("files") or \
            source_before.get("directories") != copied_snapshot.get("directories"):
        raise ValueError("copied snapshot inventory mismatch")
    if source_before.get("tree_sha256") != source_after.get("tree_sha256") or \
            source_before.get("tree_sha256") != copied_snapshot.get("tree_sha256"):
        raise ValueError("inventory tree identity mismatch")
    control_file_hashes = {}
    for item in source_before.get("files", []):
        name = str(item.get("path", "")).rsplit("/", 1)[-1]
        if name in CONTROL_NAMES:
            control_file_hashes[item["path"]] = item["sha256"]

    def visit(directory, relative_directory=""):
        names = sorted(os.listdir(directory))
        for name in names:
            path = os.path.join(directory, name)
            relative = name if not relative_directory else \
                relative_directory + "/" + name
            if relative in (excluded, MANIFEST_DIGEST_NAME,
                            COMPLETION_MARKER_NAME):
                continue
            if os.path.islink(path):
                raise ValueError("export contains symlink: {}".format(relative))
            file_stat = os.stat(path, follow_symlinks=False)
            if stat.S_ISDIR(file_stat.st_mode):
                reject_sensitive_path(relative, path, export_root)
                directories.append(relative)
                visit(path, relative)
            elif stat.S_ISREG(file_stat.st_mode):
                reject_sensitive_path(relative, path, export_root)
                files.append({
                    "path": relative,
                    "size": int(file_stat.st_size),
                    "sha256": sha256_file(path),
                })
            else:
                raise ValueError("export contains special file: {}".format(relative))

    visit(export_root)
    files.sort(key=lambda item: item["path"])
    directories.sort()
    payload = {
        "export_schema_version": 1,
        "status": "PASS",
        "schema_version": 1,
        "manifest_scope": (
            "all_export_bytes_except_manifest_and_completion_marker"
        ),
        "completion_marker": {
            "path": COMPLETION_MARKER_NAME,
            "sha256": hashlib.sha256(COMPLETION_MARKER_BYTES).hexdigest(),
            "written_last": True,
            "required": True,
        },
        "manifest_digest": {
            "path": MANIFEST_DIGEST_NAME,
            "written_after_manifest": True,
        },
        "attempt_id": metadata.get("attempt_id"),
        "tool_filename": metadata.get("tool_filename"),
        "tool_sha256": metadata.get("tool_sha256"),
        "tool_size": metadata.get("tool_size"),
        "hostname": metadata.get("hostname"),
        "collection_timestamp_utc": metadata.get("collection_timestamp_utc"),
        "expected_baseline_sha": metadata.get("expected_baseline_sha"),
        "resolved_nginx_config_paths": metadata.get(
            "resolved_nginx_config_paths"
        ),
        "resolved_nginx_config_sha256": metadata.get(
            "resolved_nginx_config_sha256"
        ),
        "resolved_served_root": metadata.get("resolved_served_root"),
        "source_before_inventory": source_before,
        "source_after_inventory": source_after,
        "copied_snapshot_inventory": copied_snapshot,
        "sidecar_summary": sidecar,
        "copied_snapshot_root_relative_path": "legacy-flat-root",
        "source_file_count": source_before.get("file_count"),
        "source_total_size": source_before.get("total_size"),
        "source_canonical_tree_sha256": source_before.get("tree_sha256"),
        "control_file_hashes": control_file_hashes,
        "sanitization": metadata.get("sanitization"),
        "identity_evidence": metadata.get("identity_evidence"),
        "metadata": metadata,
        "directories": directories,
        "files": files,
        "file_count": len(files),
        "total_size": sum(int(item["size"]) for item in files),
        "export_tree_sha256": canonical(files),
    }
    write_json(output_path, payload)
    return payload
